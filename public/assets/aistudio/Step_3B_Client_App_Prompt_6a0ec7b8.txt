rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Helper: Checks if caller is authenticated
    function isAuthenticated() {
      return request.auth != null;
    }

    // Helper: Validates admin authorization
    // Authorized strictly via custom claims, admins/{adminUid}, or users/{adminUid} admin role.
    // All optional claim/field checks guard key existence first so missing fields never abort evaluation.
    function isAdmin() {
      return isAuthenticated() && (
        ('isAdmin' in request.auth.token && request.auth.token.isAdmin == true) ||
        ('admin' in request.auth.token && request.auth.token.admin == true) ||
        ('role' in request.auth.token && request.auth.token.role == 'admin') ||
        exists(/databases/$(database)/documents/admins/$(request.auth.uid)) ||
        (exists(/databases/$(database)/documents/users/$(request.auth.uid)) && (
          ('isAdmin' in get(/databases/$(database)/documents/users/$(request.auth.uid)).data &&
            get(/databases/$(database)/documents/users/$(request.auth.uid)).data.isAdmin == true) ||
          ('role' in get(/databases/$(database)/documents/users/$(request.auth.uid)).data &&
            get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin')
        ))
      );
    }

    // Dedicated Admins registry collection
    // Deletes are strictly forbidden so admin records cannot be removed from client/SDK calls.
    match /admins/{adminUid} {
      allow get: if isAdmin() || (isAuthenticated() && request.auth.uid == adminUid);
      allow list: if isAdmin();
      allow create, update: if isAdmin();
      allow delete: if false;
    }

    // Banned Hardware registry (one-tap hardware ban enforcement)
    match /banned_hardware/{hardwareId} {
      allow read, write: if isAdmin();
    }

    // Global Application Settings (e.g., app_settings/global kill-switch & maintenance notice)
    match /app_settings/{settingId} {
      allow get: if true;
      allow list: if isAdmin();
      allow write: if isAdmin();
    }

    // Admin Settings (app_config, payment_config, security_config, server_clock)
    // Public get is allowed ONLY for app_config before sign-in; security_config and all other
    // settings require authentication; list and all writes are strictly admin-only.
    match /admin_settings/{settingId} {
      allow get: if settingId == 'app_config' || isAuthenticated();
      allow list: if isAdmin();
      allow write: if isAdmin();
    }

    // Global Messages (Broadcast Popups and System Notifications)
    match /global_messages/{messageId} {
      allow read: if isAuthenticated();
      allow write: if isAdmin();
    }

    // Users Collection: Strictly READ-ONLY for authenticated clients at users/{firebaseUid}
    // Preserves firebaseUid == request.auth.uid and uid == request.auth.uid ownership read checks.
    // Direct client writes (including points, pointsBalance, activePassType, passExpiryTimestamp,

    match /users/{userId} {
      allow read: if isAdmin() || (isAuthenticated() && (

        (resource != null && (
          ('firebaseUid' in resource.data && resource.data.firebaseUid == request.auth.uid) ||
          ('uid' in resource.data && resource.data.uid == request.auth.uid)
        ))
      ));
      allow write: if isAdmin();

      // Direct User Messages Subcollection (Broadcast / Rejection notices)
      match /messages/{messageId} {
        allow read, delete: if isAdmin() || (isAuthenticated() && request.auth.uid == userId);
        allow create, update: if isAdmin();
      }

      // Referral Audit Records (read-only for owner, written exclusively by Admin transactions)
      match /referrals/{refereeUid} {
        allow read: if isAdmin() || (isAuthenticated() && request.auth.uid == userId);
        allow write: if isAdmin();
      }

      // Admin Points Audit Subcollection (Strictly Admin-only)
      match /points_audit/{logId} {
        allow read, write: if isAdmin();
      }
    }

    // Helper: Validates server-backed rate limiting for pending actions (12s cooldown per user)
    // Safely supports both Firestore timestamp and numeric epoch-millis lastActionTime values.
    function isWithinPendingActionRateLimit() {
      let userDoc = /databases/$(database)/documents/users/$(request.auth.uid);
      return !exists(userDoc) ||
             !('lastActionTime' in get(userDoc).data) ||
             (get(userDoc).data.lastActionTime is timestamp &&
               request.time >= get(userDoc).data.lastActionTime + duration.value(12, 's')) ||
             ((get(userDoc).data.lastActionTime is int || get(userDoc).data.lastActionTime is float) &&
               request.time.toMillis() >= get(userDoc).data.lastActionTime + 12000) ||
             (!(get(userDoc).data.lastActionTime is timestamp) &&
               !(get(userDoc).data.lastActionTime is int) &&
               !(get(userDoc).data.lastActionTime is float));
    }

    // Helper: Enforces flat canonical pending_actions key allowlist and blocks nested/privileged fields
    function hasValidPendingActionSchema(data) {

        'oldEmail', 'old_email', 'previousEmail',
        'oldFirebaseUid', 'old_uid', 'previousUid',
        'legacyCustomUserId', 'oldDocumentId', 'old_document_id',
        'previousDocumentId', 'migratedFromDocId',
        'referralCode', 'code',
        'passTier', 'passType', 'pass_type',
        'utr', 'transactionId', 'paymentRef', 'refNo',
        'amount', 'paymentAmount',
        'featureId', 'featureKey', 'featureName', 'feature', 'featureIds', 'hours',
        'pointsCost', 'durationDays', 'durationMs', 'condition',
        'autoAcceptEnabled', 'isAutoAcceptEnabled',

        'isMaxPickupDistanceEnabled', 'maxPickupDistance',
        'isMaxDropDistanceEnabled', 'maxDropDistance',

                                       !('data' in data) &&
                                       !('email' in data) &&
                                       !('isAdmin' in data) &&
                                       !('role' in data) &&
                                       !('points' in data) &&
                                       !('pointsBalance' in data) &&
                                       !('newPointsBalance' in data) &&
                                       !('pointsAdded' in data) &&
                                       !('pointsDeducted' in data) &&
                                       !('activePassType' in data) &&
                                       !('passExpiryTimestamp' in data) &&
                                       !('verifiedUtr' in data) &&
                                       !('processedBy' in data) &&
                                       !('processedAt' in data) &&
                                       !('completedAt' in data) &&
                                       !('failedAt' in data) &&
                                       !('errorMessage' in data) &&
                                       !('resolvedUserId' in data) &&
                                       !('banUntil' in data) &&
                                       !('banUntilTimestamp' in data) &&
                                       !('isBanned' in data) &&
                                       !('hardwareBanned' in data) &&
                                       !('isPaymentVerified' in data) &&
                                       !('locked_features' in data) &&
                                       !('lockedFeatures' in data) &&
                                       !('unlocked_features' in data) &&
                                       !('unlockedFeatures' in data) &&
                                       !('featureExpiries' in data) &&
                                       !('trialExpiry' in data) &&

    }

    // Helper: Enforces flat canonical parameter types and numeric bounds on pending_actions
    function hasValidPendingActionFields(data) {
      let hasValidFlatTypes = (!('userEmail' in data) || data.userEmail is string) &&
                              (!('passTier' in data) || data.passTier is string) &&
                              (!('utr' in data) || data.utr is string) &&
                              (!('referralCode' in data) || data.referralCode is string) &&
                              (!('hardwareId' in data) || data.hardwareId is string) &&
                              (!('linkedHardwareId' in data) || data.linkedHardwareId is string) &&
                              (!('featureId' in data) || data.featureId is string) &&
                              (!('featureIds' in data) || data.featureIds is list) &&
                              (!('hours' in data) || data.hours is int || data.hours is float) &&
                              (!('currentFolder' in data) || data.currentFolder is string) &&
                              (!('current_folder' in data) || data.current_folder is string);
      let hasValidPointsCost = !('pointsCost' in data) || (
        (data.pointsCost is int || data.pointsCost is float) &&
        data.pointsCost >= 0 &&
        data.pointsCost <= 100000
      );
      let hasValidDurationMs = !('durationMs' in data) || (
        (data.durationMs is int || data.durationMs is float) &&
        data.durationMs >= 0 &&

        data.durationDays >= 0 &&
        data.durationDays <= 3650
      );
      let hasValidCondition = !('condition' in data) || (data.condition is string || data.condition is map);
      return hasValidFlatTypes &&
             hasValidPointsCost &&
             hasValidDurationMs &&
             hasValidDurationDays &&
             hasValidCondition;
    }

    // Helper: Strict Zero-Trust validation for client-enqueued pending actions (Flat Canonical Contract)
    function isValidPendingAction() {
      let data = request.resource.data;
      let hasCanonicalOwner = ('uid' in data) &&
                              data.uid is string &&
                              data.uid == request.auth.uid &&
                              (!('firebaseUid' in data) || (data.firebaseUid is string && data.firebaseUid == request.auth.uid)) &&
                              (!('userId' in data) || data.userId is string && data.userId == request.auth.uid);
      let hasPendingStatus = ('status' in data) && data.status == 'PENDING';
      let hasAllowedType = ('actionType' in data) &&
                           data.actionType is string &&
                           data.actionType.size() > 0 &&
                           data.actionType != 'ADD_POINTS' &&
                           data.actionType in [
                             'INIT_PROFILE',
                             'UPDATE_PROFILE',
                             'MIGRATE_PROFILE',
                             'CLAIM_REFERRAL',
                             'PURCHASE_PASS',
                             'ACTIVATE_PASS_PAYMENT',
                             'UNLOCK_FEATURE',
                             'UNLOCK_MULTIPLE_FEATURES',
                             'CLAIM_TRIAL',
                             'REWARD_AD_WATCHED'
                           ];
      let noRewardAmountOverride = data.actionType != 'REWARD_AD_WATCHED' || !('amount' in data);
      let hasValidTimestamp = ('createdAt' in data) &&
                              data.createdAt is timestamp &&
                              data.createdAt <= request.time;
      return hasCanonicalOwner &&
             hasPendingStatus &&
             hasAllowedType &&
             noRewardAmountOverride &&
             hasValidTimestamp &&
             hasValidPendingActionSchema(data) &&
             hasValidPendingActionFields(data);
    }

    // Helper: Strict Zero-Trust validation for client UTR verification submissions
    function isValidPendingVerificationCreate() {
      let data = request.resource.data;
      let hasOwner = (('uid' in data && data.uid == request.auth.uid) ||
                      ('firebaseUid' in data && data.firebaseUid == request.auth.uid)) &&
                     (!('userId' in data) || data.userId == request.auth.uid);
      let hasAllowedKeys = data.keys().hasOnly([
        'deviceHardwareId', 'hardwareId', 'passName', 'passTier', 'passType',
        'amount', 'paymentAmount', 'utr', 'status', 'isUsed', 'userId',
        'firebaseUid', 'uid', 'userEmail', 'email', 'timestamp', 'createdAt', 'submittedAt', 'notes'
      ]);
      let hasValidCoreFields = ('status' in data) && data.status == 'PENDING' &&
                               ('isUsed' in data) && data.isUsed == false &&
                               ('utr' in data) && data.utr is string &&
                               data.utr.matches('^[0-9]{12}$') &&
                               (!('amount' in data) || data.amount is int || data.amount is float) &&
                               (!('deviceHardwareId' in data) || data.deviceHardwareId is string) &&
                               (!('passName' in data) || data.passName is string) &&
                               (!('passTier' in data) || data.passTier is string) &&
                               (!('passType' in data) || data.passType is string) &&
                               (!('timestamp' in data) || data.timestamp is timestamp || data.timestamp is int || data.timestamp is float) &&
                               (!('createdAt' in data) || data.createdAt is timestamp || data.createdAt is int || data.createdAt is float);
      return hasOwner && hasAllowedKeys && hasValidCoreFields;
    }

    // Pending Actions (Zero-Trust Flat Canonical Client Queue processed atomically by Admin)
    match /pending_actions/{actionId} {
      allow create: if isAdmin() || (
        isAuthenticated() &&
        isValidPendingAction() &&
        isWithinPendingActionRateLimit()
      );
      allow read: if isAuthenticated() && (
        isAdmin() ||
        resource.data.uid == request.auth.uid
      );
      allow update, delete: if isAdmin();
    }

    // Pending Verifications (Client UTR submission queue; immutable after creation by client)
    match /pending_verifications/{utr} {
      allow create: if isAdmin() || (
        isAuthenticated() &&
        utr.matches('^[0-9]{12}$') &&
        request.resource.data.utr == utr &&
        isValidPendingVerificationCreate()
      );
      allow read: if isAdmin() || (
        isAuthenticated() && (
          ('uid' in resource.data && resource.data.uid == request.auth.uid) ||
          ('firebaseUid' in resource.data && resource.data.firebaseUid == request.auth.uid) ||
          ('userId' in resource.data && resource.data.userId == request.auth.uid)
        )
      );
      allow update: if isAdmin();
      allow delete: if isAdmin() || (
        isAuthenticated() && (
          ('uid' in resource.data && resource.data.uid == request.auth.uid) ||
          ('firebaseUid' in resource.data && resource.data.firebaseUid == request.auth.uid) ||
          ('userId' in resource.data && resource.data.userId == request.auth.uid)
        ) &&
        resource.data.status == 'PENDING'
      );
    }

    // Received Payments Ledger (Strictly Admin-only)
    match /received_payments/{utr} {
      allow read, write: if isAdmin();
    }

    // Fallback: Disallow all other collections by default
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
